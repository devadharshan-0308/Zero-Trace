WipeService.kt
package com.example.securewipecompanion

import android.app.Service
import android.content.Intent
import android.net.Uri
import android.os.IBinder
import android.util.Log
import kotlinx.coroutines.*
import org.json.JSONObject
import java.io.*
import java.net.ServerSocket
import java.net.Socket
import java.security.SecureRandom

class WipeService : Service() {

    private val SERVER_PORT = 5555
    private var serverSocket: ServerSocket? = null
    private val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    private val secureRandom = SecureRandom()

    companion object {
        private const val TAG = "WipeService"
        private const val WIPE_PASSES = 3
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        Log.i(TAG, "WipeService started in foreground")
        return START_STICKY
    }

    private suspend fun startServer() {
        try {
            serverSocket = ServerSocket(SERVER_PORT)
            Log.i(TAG, "✅ Server listening on port $SERVER_PORT")

            while (serviceScope.isActive) {
                val client = serverSocket!!.accept()
                Log.i(TAG, "🔗 Client connected: ${client.inetAddress}")
                handleClient(client)
            }
        } catch (e: Exception) {
            Log.e(TAG, "❌ Server error: ${e.message}")
        }
    }

    private fun handleClient(client: Socket) {
        serviceScope.launch {
            try {
                val reader = BufferedReader(InputStreamReader(client.getInputStream()))
                val writer = PrintWriter(OutputStreamWriter(client.getOutputStream()), true)

                val request = reader.readLine()
                Log.i(TAG, "📨 Received request: $request")

                val json = JSONObject(request)
                val response = JSONObject()

                when (json.getString("action")) {
                    "ping" -> {
                        response.put("status", "success")
                        response.put("message", "WipeService is running and ready")
                    }
                    "wipe" -> {
                        val path = json.optString("path", null)
                        val uriStr = json.optString("uri", null)

                        val success = when {
                            path != null -> secureWipeFile(File(path))
                            uriStr != null -> secureWipeUri(Uri.parse(uriStr))
                            else -> {
                                response.put("message", "No file path or URI provided")
                                false
                            }
                        }

                        if (success) {
                            response.put("status", "success")
                            response.put("message", "File securely wiped and deleted")
                        } else {
                            response.put("status", "error")
                            response.put("message", "Failed to wipe file - check permissions or file existence")
                        }
                    }
                    else -> {
                        response.put("status", "error")
                        response.put("message", "Unknown action. Use 'ping' or 'wipe'")
                    }
                }

                writer.println(response.toString())
                Log.i(TAG, "📤 Sent response: ${response.toString()}")

                client.close()
            } catch (e: Exception) {
                Log.e(TAG, "❌ Client handling error: ${e.message}")
            }
        }
    }

    private fun secureWipeFile(file: File): Boolean {
        return try {
            Log.i(TAG, "🔒 Starting secure wipe for: ${file.absolutePath}")

            if (!file.exists()) {
                Log.e(TAG, "❌ File does not exist: ${file.absolutePath}")
                return false
            }

            if (!file.canWrite()) {
                Log.e(TAG, "❌ No write permission for file: ${file.absolutePath}")
                return false
            }

            val size = file.length()
            Log.i(TAG, "📊 File size: $size bytes")

            // Multiple pass wiping for better security
            repeat(WIPE_PASSES) { pass ->
                Log.i(TAG, "🔄 Wipe pass ${pass + 1}/$WIPE_PASSES")

                var remaining = size
                val fos = FileOutputStream(file)

                while (remaining > 0) {
                    val buffer = when (pass) {
                        0 -> ByteArray(4096) { 0x00 } // Zero pass
                        1 -> ByteArray(4096) { 0xFF.toByte() } // Ones pass
                        else -> { // Random pass
                            ByteArray(4096).also { secureRandom.nextBytes(it) }
                        }
                    }

                    val bytesToWrite = minOf(buffer.size.toLong(), remaining).toInt()
                    fos.write(buffer, 0, bytesToWrite)
                    remaining -= bytesToWrite
                }

                fos.flush()
                fos.fd.sync() // Force sync to disk
                fos.close()
            }

            // Final deletion
            val deleteSuccess = file.delete()
            Log.i(TAG, "🗑️ File deletion: $deleteSuccess")

            if (deleteSuccess) {
                Log.i(TAG, "✅ File securely wiped and deleted: ${file.absolutePath}")
                true
            } else {
                Log.e(TAG, "❌ Failed to delete file after wiping: ${file.absolutePath}")
                false
            }
        } catch (e: Exception) {
            Log.e(TAG, "❌ Error wiping file: ${e.message}")
            false
        }
    }

    private fun secureWipeUri(uri: Uri): Boolean {
        return try {
            Log.i(TAG, "🔒 Starting secure wipe for URI: $uri")

            val pfd = contentResolver.openFileDescriptor(uri, "rw") ?: run {
                Log.e(TAG, "❌ Cannot open file descriptor for URI: $uri")
                return false
            }

            val size = pfd.statSize
            Log.i(TAG, "📊 URI file size: $size bytes")

            // Multiple pass wiping
            repeat(WIPE_PASSES) { pass ->
                Log.i(TAG, "🔄 URI wipe pass ${pass + 1}/$WIPE_PASSES")

                var remaining = size
                val fos = FileOutputStream(pfd.fileDescriptor)

                while (remaining > 0) {
                    val buffer = when (pass) {
                        0 -> ByteArray(4096) { 0x00 }
                        1 -> ByteArray(4096) { 0xFF.toByte() }
                        else -> ByteArray(4096).also { secureRandom.nextBytes(it) }
                    }

                    val bytesToWrite = minOf(buffer.size.toLong(), remaining).toInt()
                    fos.write(buffer, 0, bytesToWrite)
                    remaining -= bytesToWrite
                }

                fos.flush()
                fos.fd.sync()
                fos.close()
            }

            pfd.close()

            // Try to delete the file through content resolver
            val deleteSuccess = try {
                contentResolver.delete(uri, null, null) > 0
            } catch (e: Exception) {
                Log.w(TAG, "⚠️ Cannot delete via content resolver, file wiped but not deleted")
                true // Consider success if we wiped it even if we can't delete
            }

            Log.i(TAG, "✅ URI securely wiped: $uri, deleted: $deleteSuccess")
            deleteSuccess
        } catch (e: Exception) {
            Log.e(TAG, "❌ Error wiping URI: ${e.message}")
            false
        }
    }

    override fun onCreate() {
        super.onCreate()
        serviceScope.launch { startServer() }
        Log.i(TAG, "✅ WipeService created and server started")
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onDestroy() {
        super.onDestroy()
        serverSocket?.close()
        serviceScope.cancel()
        Log.i(TAG, "🛑 WipeService destroyed")
    }
