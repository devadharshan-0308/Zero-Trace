import subprocess
import socket
import json
import time
import sys

ADB_PATH = "adb"
ANDROID_SERVICE_PORT = 5555

class AndroidBridge:
    def __init__(self):
        self.device_id = None

    def detect_device(self):
        """Detect connected Android device"""
        try:
            result = subprocess.run([ADB_PATH, "devices"], capture_output=True, text=True, check=True)
            lines = result.stdout.strip().split('\n')
            for line in lines[1:]:
                if line.strip() and "device" in line:
                    self.device_id = line.split('\t')[0]
                    print(f"✅ Device detected: {self.device_id}")
                    return self.device_id
            print("❌ No Android device detected")
            return None
        except subprocess.CalledProcessError as e:
            print(f"❌ ADB error: {e}")
            return None
        except FileNotFoundError:
            print("❌ ADB not found. Make sure Android Platform Tools are installed and in PATH")
            return None

    def forward_port(self):
        """Forward TCP port from device to local machine"""
        try:
            # Remove existing forward first
            subprocess.run([
                ADB_PATH, "-s", self.device_id, "forward",
                "--remove", f"tcp:{ANDROID_SERVICE_PORT}"
            ], capture_output=True)
            
            time.sleep(1)
            
            # Create new forward
            result = subprocess.run([
                ADB_PATH, "-s", self.device_id, "forward",
                f"tcp:{ANDROID_SERVICE_PORT}", f"tcp:{ANDROID_SERVICE_PORT}"
            ], capture_output=True, text=True, check=True)
            
            print(f"✅ Port forwarded: localhost:{ANDROID_SERVICE_PORT} → device:{ANDROID_SERVICE_PORT}")
            return True
        except subprocess.CalledProcessError as e:
            print(f"❌ Port forwarding failed: {e}")
            return False

    def send_command(self, payload):
        """Send command to Android service"""
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(10)
        try:
            print(f"🔗 Connecting to service...")
            s.connect(('127.0.0.1', ANDROID_SERVICE_PORT))
            print(f"📤 Sending command: {payload}")
            
            data = json.dumps(payload).encode('utf-8')
            s.sendall(data + b'\n')
            
            response = s.recv(4096)
            result = json.loads(response.decode('utf-8'))
            print(f"📥 Response received: {result}")
            return result
        except socket.timeout:
            print("⏰ Connection timeout")
            return {"status": "error", "message": "Connection timeout"}
        except ConnectionRefusedError:
            print("❌ Connection refused - service may not be running")
            return {"status": "error", "message": "Connection refused - service not running"}
        except json.JSONDecodeError as e:
            print(f"❌ Invalid JSON response: {e}")
            return {"status": "error", "message": "Invalid response from service"}
        except Exception as e:
            print(f"❌ Communication error: {e}")
            return {"status": "error", "message": str(e)}
        finally:
            s.close()

    def test_connection(self):
        """Test if the service is responding"""
        print("🔄 Testing connection to WipeService...")
        result = self.send_command({"action": "ping"})
        return result.get("status") == "success"

    def wipe_file(self, file_path):
        """Wipe a specific file"""
        print(f"🗑️ Attempting to wipe file: {file_path}")
        result = self.send_command({"action": "wipe", "path": file_path})
        return result

    def get_app_info(self):
        """Get information about the app's file directory"""
        try:
            # Get the app's external files directory
            result = subprocess.run([
                ADB_PATH, "-s", self.device_id, "shell",
                "run-as", "com.example.securewipecompanion",
                "ls", "-la", "/data/data/com.example.securewipecompanion/files/"
            ], capture_output=True, text=True)
            
            if result.returncode == 0:
                print("📁 App files directory:")
                print(result.stdout)
            return result.stdout
        except Exception as e:
            print(f"❌ Error getting app info: {e}")
            return None

def main():
    bridge = AndroidBridge()
    
    print("=" * 50)
    print("🔒 SECURE WIPE COMPANION - ANDROID BRIDGE")
    print("=" * 50)
    
    # Step 1: Detect device
    if not bridge.detect_device():
        print("\n💡 Troubleshooting tips:")
        print("   1. Enable USB debugging on your Android device")
        print("   2. Connect device via USB cable")
        print("   3. Authorize the computer when prompted on device")
        print("   4. Ensure ADB is in your PATH")
        sys.exit(1)
    
    # Step 2: Forward port
    if not bridge.forward_port():
        print("❌ Failed to forward port")
        sys.exit(1)
    
    # Wait for connection
    time.sleep(2)
    
    # Step 3: Test connection
    if not bridge.test_connection():
        print("\n❌ Cannot connect to WipeService")
        print("\n💡 Please ensure:")
        print("   1. The SecureWipeCompanion app is installed and OPEN")
        print("   2. You clicked 'Start WipeService (Server)' in the app")
        print("   3. No firewall is blocking the connection")
        
        # Try to start the app
        try:
            print("\n🔄 Attempting to start app...")
            subprocess.run([
                ADB_PATH, "-s", bridge.device_id, "shell",
                "am", "start", "-n", "com.example.securewipecompanion/.MainActivity"
            ], capture_output=True)
            print("✅ App started. Please click 'Start WipeService' and run this script again.")
        except:
            print("❌ Could not start app automatically")
        
        sys.exit(1)
    
    print("✅ Service connection successful!")
    
    # Step 4: Wipe test files
    print("\n" + "=" * 30)
    print("🧪 TESTING FILE WIPING")
    print("=" * 30)
    
    # Common test file paths
    test_paths = [
        "/sdcard/Android/data/com.example.securewipecompanion/files/test.txt",
        "/storage/emulated/0/Android/data/com.example.securewipecompanion/files/test.txt",
    ]
    
    success = False
    for test_path in test_paths:
        print(f"\n🔍 Trying path: {test_path}")
        result = bridge.wipe_file(test_path)
        
        if result.get("status") == "success":
            print("🎉 SUCCESS: File securely wiped and deleted!")
            print(f"   Message: {result.get('message')}")
            success = True
            break
        else:
            print(f"❌ Failed: {result.get('message', 'Unknown error')}")
    
    if not success:
        print("\n💡 All automatic paths failed. Manual steps:")
        print("   1. Check the app screen for the exact file path")
        print("   2. Use the file picker in the app to select a file")
        print("   3. The path will be displayed on screen")
        print("   4. Update the test_paths in the script with the correct path")
    
    print("\n" + "=" * 50)
    print("✅ Script execution completed")
    print("=" * 50)

if __name__ == "__main__":
    main()
